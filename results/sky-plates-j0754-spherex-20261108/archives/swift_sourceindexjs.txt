$(document).ready(reset);

function reset()
{
  
  topurl=$('#topurl').html();
  sourceID=$('#sid').html();
  stem=$('#psid').html();
  scrollLinks();
  $('#stab_controls').submit(updateStab);
  updateStab();
  updateDetTable();
  updateUndetTable(); 
  updatexCorrTable();
  changeStamp(0);
//   $('.stamp').draggable();
  setupHelp();
  setupLC();
}

function changeStamp(what)
{
  var imstem=stem+"_"+images[what];
  
  for (a=0; a<5; a++)
  {
    if (a==what)
      $('#stamplink'+a).css({'fontWeight':'bold', 'color':'black', 'textDecoration':'none'});
    else
    {
      $('#stamplink'+a).css({'fontWeight':'normal', 'color':'blue', 'textDecoration':'underline'});
      $('#stampimlink'+a).css('display', 'none');
    }
  }
  $('#stampimlink'+what).css('display', 'block');
}

function showStab()
{
  $("#stab_controls").slideDown("fast");
  $('#stab_ctrl_link').attr('href', "javascript:hideStab();").html("Hide table controls");
}

function hideStab()
{
  $("#stab_controls").slideUp("fast");
  $('#stab_ctrl_link').attr('href', "javascript:showStab();").html("Show table controls");;
}

function updateStab(event=-1)
{
  if (event!=-1)
    event.preventDefault();
  
  $('#stabWork').css('display','block');
  var formVals=$('#stab_controls').serialize();
  
  $('#stab_div').load(topurl+'/getSourceSpecDetailsTable.php', formVals, function()
    {
     $('#stabWork').css('display','none');
     setupHelp();
    });
  
 
}

function updateDetTable()

}

function sortxCorr(what)
{
  var xCorrCol=$('#xctab_sortCol').val();
  var xCorrDir=$('#xctab_sortDir').val();
    
  if (what==xCorrCol)
    xCorrDir=1-xCorrDir;
  else
    xCorrDir=0;
  
  xCorrCol=what;
    
  $('#xctab_sortCol').val(xCorrCol);
  $('#xctab_sortDir').val(xCorrDir);
    
  updatexCorrTable();
}

function showAllRows(label,num)
{
  $('#'+label+'_extra').fadeIn('slow');
  $('#'+label+'_showAll').val(1);
  $('#'+label+'_stacked_showAll').val(1);
  $('#'+label+'_showextra').html("Showing all "+num+" rows. <a href=\"javascript:hideAllRows('"+label+"',"+num+");\">Show only the first 10</a>.");
}



function hideAllRows(label,num)
{
  $('#'+label+'_extra').fadeOut('slow');
  $('#'+label+'_showAll').val(0);
  $('#'+label+'_stacked_showAll').val(0);
  $('#'+label+'_showextra').html("Showing only the first 10 (/"+num+") rows. <a href=\"javascript:showAllRows('"+label+"',"+num+");\">Show all</a>.");
}


function showDetDetails(obs,band)
{
  var $headTD=$('#'+obs+' > td:first');
  
  $headTD.attr('rowspan',parseInt($headTD.attr('rowspan'))+1);
  $('#detDetails_'+obs+'_'+band).fadeIn(1000);
  
  var $prevCell=$('#detDetailsLink_'+obs+'_'+band);
  $prevCell.attr('rowspan',2);
  
  var theLink=$prevCell.find('a.small');
  theLink.html("hide details");
  theLink.attr('href', 'javascript:hideDetDetails("'+obs+'",'+band+');');

 
}

function hideDetDetails(obs,band)
{
  $('#detDetails_'+obs+'_'+band).fadeOut('fast');
  var $headTD=$('#'+obs+' > td:first');
  $headTD.attr('rowspan',parseInt($headTD.attr('rowspan'))-1);
  var $prevCell=$('#detDetailsLink_'+obs+'_'+band);
  $prevCell.attr('rowspan',1);
  
  
  var theLink=$prevCell.find('a.small');
  theLink.html("details");
  theLink.attr('href', 'javascript:showDetDetails("'+obs+'",'+band+');');

 
}
