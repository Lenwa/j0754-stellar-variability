function getLeftRight(gl, centerWave) {
	var center = gl.toDomCoords(centerWave, 0);
	var left = center[0] - 0.5;
	var right = center[0] + 0.5;
	return [ left, right ];
}

function getClosestValues(a, x) {
	var lo = 0, hi = a.length;
	while (hi - lo > 1) {
		var mid = (lo + hi) / 2;
		if (a[mid] <= x) {
			lo = mid;
		} else {
			hi = mid;
		}
	}
	if (a[lo] == x)
		hi = lo;
	return [ a[lo], a[hi] ];
}

function cleanChart(speccharts){
	if(null==speccharts)return;
	if(null==speccharts.series)return;
	seriesLength = speccharts.series.length;
	if(0==seriesLength) return;
	for(var i = seriesLength - 1; i >= 0; i--) {
		speccharts.series[i].remove(false);
	}
	speccharts.redraw();
}

    				sr=getSmoothedSerie(spectrum_r, item, colors[colorindex_r++]);
    				spectrum_r.addSeries(sr, false);
    			}
    		}
    	}
    	spectrum_b.redraw();
    	spectrum_r.redraw();
    },"json");

}
function getSerie(speccharts, json, color){
	json.data.sort();
	sr = {
			name: json.extname,
			data: json.data,
			type: 'line',
			color: color,
			yAxis: 0,
			marker: {enabled: false},
			id:speccharts.title+'_data_'+json.extname,
			//turboThreshold:json.data.length
			events: {
				click:function(){
                	var groupName = json.extname;
                	var pg = $('#obj_info');
                	pg.propertygrid('collapseGroup');
                	var groups = pg.propertygrid('groups');
                	var index = $.inArray(groupName, $.map(groups, function(g){return g.value}));
                	pg.propertygrid('expandGroup', index);
				},
                mouseOver: function () {
                },
                mouseOut: function () {
                }
            }
		};
	return sr;
}
function getSmoothedSerie(speccharts, json, color){
	json.data.sort();
	smoothwin = parseInt($('#smoothwindow').val());	
	sr = {
			name: json.extname+"-smoothed",
			data: smoothArray(json.data, smoothwin),
			type: 'line',
			color: color,
			yAxis: 0,
			marker: {enabled: false},
			linkedTo: speccharts.title+'_data_'+json.extname,
			id:speccharts.title+'_smoothed_'+json.extname,
			//turboThreshold:json.data.length
		};
	return sr;
}

function smoothArray(array, smoothing) {
	newarray = [];
	arrlen = array.length;
    for (i = 0; i < arrlen; i++) {
        sum = .0; count = 0;
        for (index = i - smoothing; index <= i + smoothing; index++) {
        	if(index<0 || index>=arrlen) continue;
        	thei = index;//index < 0 ? index + arrlen : index % arrlen;
            sum += array[thei][1];
            count++;
        }
        
        newarray[i]=[];
        newarray[i][0] = array[i][0];
        newarray[i][1] = sum/count;//sum/((smoothing*2)+1);
        newarray[i][2] = array[i][2];
    }
    return newarray;
}
function getInfo(obsid){	
	$.post('../medspectrum/info/'+obsid,{},function(json,status){
		rows=json.rows;
		$.each(rows, function(index, row){
			if(row['lmjm']==null || row['lmjm']=='None') {
				groupname='COADD_'+row['band'];
			}else{
				groupname=row['band']+'-'+row['lmjm'];
			}
			$.each(row, function(key, val){
				var row = {name:key, value:val, group:groupname};
				$('#obj_info').propertygrid('appendRow',row);
			});
		});
		$('#obj_info').propertygrid('collapseGroup');
		$('#obj_info').propertygrid('expandGroup', 0);
	});	
}

			$('#nearby_objs').propertygrid('appendRow',row);
		});
	});	
}

var bestState = {obsid:null, loaded:null, loading:null, seq:0, rows:[]};

function bestNumSorter(a, b){
	var na = parseFloat(a), nb = parseFloat(b);
	var aBad = !isFinite(na), bBad = !isFinite(nb);
	if(aBad && bBad) return 0;
	if(aBad) return -1;
	if(bBad) return 1;
	return na < nb ? -1 : (na > nb ? 1 : 0);
}

function bestNumFormatter(value){
	return formatBestNum(value);
}

function formatBestNum(value){
	if(value === null || value === undefined || value === '') return '';
	var number = Number(value);
	if(!isFinite(number)) return value;
	return number.toFixed(4);
}

function showBestMessage(text){
	if(text){
		$('#best_msg').text(text).show();
	}else{
		$('#best_msg').hide().text('');
	}
	try{ $('#best_objs').datagrid('resize'); }catch(e){}
}

function showBestMatch(data){
	var sourceId = (data && data.source_id) ? String(data.source_id) : '-';
	var sepText = '-';
	if(data && data.sep_arcsec !== null && data.sep_arcsec !== undefined && data.sep_arcsec !== ''){
		var number = Number(data.sep_arcsec);
		sepText = isFinite(number) ? (number.toFixed(4) + '″') : String(data.sep_arcsec);
	}
	$('#best_source_id').text(sourceId);
	$('#best_sep').text(sepText);
}

function bestFieldText(value){
	if(value === null || value === undefined) return '';
	if(typeof value === 'number') return formatBestNum(value).toLowerCase();
	return String(value).toLowerCase();
}

function applyBestFilter(){
	var group = ($('#best_filter_group').val() || '').toLowerCase();
	var band = ($('#best_filter_band').val() || '').toLowerCase();
	var mag = ($('#best_filter_mag').val() || '').toLowerCase();
	var err = ($('#best_filter_err').val() || '').toLowerCase();
	var rows = [];
	$.each(bestState.rows || [], function(index, row){
		if(group && bestFieldText(row.group).indexOf(group) < 0) return;
		if(band && bestFieldText(row.band).indexOf(band) < 0) return;
		if(mag && bestFieldText(row.mag).indexOf(mag) < 0) return;
		if(err && bestFieldText(row.err).indexOf(err) < 0) return;
		rows.push(row);
	});
	var grid = $('#best_objs');
	var opts = grid.datagrid('options');
	if(opts.sortName){
		var field = opts.sortName;
		var direction = opts.sortOrder === 'desc' ? -1 : 1;
		var sorter = (field === 'mag' || field === 'err') ? bestNumSorter : function(a, b){
			return String(a == null ? '' : a).localeCompare(String(b == null ? '' : b));
		};
		rows.sort(function(a, b){ return sorter(a[field], b[field]) * direction; });
	}
	grid.datagrid('loadData', {rows:rows});
}

function renderBest(data){
	showBestMatch(data);
	var rows = [];
	$.each(data.rows || [], function(index, row){
		rows.push({group:row.group, band:row.band, mag:row.mag, err:row.err});
	});
	bestState.rows = rows;
	if(!rows.length){
		$('#best_objs').datagrid('loadData', {rows:[]});
		showBestMessage('No BEST data');
		return;
	}
	showBestMessage('');
	applyBestFilter();
}

function bestPanelOpen(){
	var tabs = $('#info-tabs');
	if(!tabs.length || !tabs.data('tabs')) return false;
	var selected = tabs.tabs('getSelected');
	return selected && selected.panel('options').title === 'BEST';
}

function onInfoTabSelect(title){
	var gridId = title === 'Object' ? '#obj_info' : (title === 'Nearby' ? '#nearby_objs' : '#best_objs');
	setTimeout(function(){
		try{
			var grid = $(gridId);
			if(grid.data('datagrid')) grid.datagrid('resize');
			else grid.propertygrid('resize');
		}catch(e){}
	}, 0);
	if(title === 'BEST') loadBestPhot();
}

function resetBest(obsid){
	bestState.seq += 1;
	bestState.obsid = obsid;
	bestState.loaded = null;
	bestState.loading = null;
	bestState.rows = [];
	$('#best_objs').datagrid('loadData', {rows:[]});
	showBestMatch(null);
	showBestMessage('');
	if(bestPanelOpen()) loadBestPhot();
}

function loadBestPhot(){
	var obsid = bestState.obsid;
	if(obsid === null || obsid === undefined || obsid === '') return;
	if(!bestPanelOpen()) return;
	if(bestState.loaded === obsid || bestState.loading === obsid) return;
	var seq = bestState.seq;
	bestState.loading = obsid;
	showBestMatch(null);
	showBestMessage('Loading');
	$.post('../medspectrum/best/'+obsid, {}, function(data){
		if(bestState.seq !== seq || bestState.obsid !== obsid) return;
		bestState.loading = null;
		bestState.loaded = obsid;
		renderBest(data);
	}).fail(function(){
		if(bestState.seq !== seq || bestState.obsid !== obsid) return;
		bestState.loading = null;
		showBestMessage('Failed to load');
	});
}

function map2text(map){
	text='';
	$.each(map, function(item){
		text+=item + ':' + map[item]+'\n';
	});
	return text;
}

function text2map(text){
	arr=text.split('\n');
	map={};
	$.each(arr, function(i, item){
		item=item.trim();
		if(item == '') return;
		ki = item.split(':');
		if(ki.length<2) return;
		map[ki[0]]=ki[1].trim();
	});
	return map;
}

function plotSpectraLines(specchart, lines, color,showline){	
	xa = specchart.xAxis[0];
	$.each(lines, function(index, val){
		realval=index;
		realname=val;
		lineid=specchart.title+'_line-'+realval+'-'+color;
		xa.removePlotLine(lineid);
		if(showline){
			var shiftedWave = realval * (1 + changez);
			addSpecLines(specchart, lineid, realname, shiftedWave, color);
		}
	});
}
function addSpecLines(specchart, lineid, name, value, color){
	line = {
			color: color, // Color value
		    dashStyle: 'solid', // Style of the plot line. Default to solid
		    value: value, // Value of where the line will appear
		    width: 1, // Width of the line
		    label:{
		    	text:name,
		    	verticalAlign:'top',
		    	rotation:0
		    },
		    id:lineid
	    };
	specchart.xAxis[0].addPlotLine(line);
}

var layoutSize = null;
var layoutResizeTimer = null;
var spectrumResizeTimer = null;
function resize() {
	width =$(window).width();
	height=$(window).height()-70;
	var $layout = $('#main-layout');
	if ($layout.children('.layout-panel:animated,.layout-expand:animated').length) {
		clearTimeout(layoutResizeTimer);
		layoutResizeTimer = setTimeout(resize, 50);
		return;
	}
	if (layoutSize && layoutSize.w === width && layoutSize.h === height) {
		return;
	}
	layoutSize = {w: width, h: height};
	$layout.layout('resize',{height:height});
}
function resizeSpectrumLayout(){
	var $layout = $('#spectrum-layout');
	if (!$layout.length) return;
	if ($layout.children('.layout-panel:animated,.layout-expand:animated').length) {
		clearTimeout(spectrumResizeTimer);
		spectrumResizeTimer = setTimeout(resizeSpectrumLayout, 50);
		return;
	}
	$layout.layout('resize');
}
